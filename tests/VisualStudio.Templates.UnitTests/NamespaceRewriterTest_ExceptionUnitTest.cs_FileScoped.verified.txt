namespace PosInformatique.VisualStudio.Templates.NamespaceTest;

public class TheClass
{
    [Fact]
    public void Constructor()
    {
        var exception = new TheClassUnderTest();

        exception.Message.Should().Be("Exception of type 'PosInformatique.VisualStudio.Templates.NamespaceUnderTest.TheClassUnderTest' was thrown.");
        exception.InnerException.Should().BeNull();
    }

    [Fact]
    public void Constructor_WithMessage()
    {
        var exception = new TheClassUnderTest("The message");

        exception.Message.Should().Be("The message");
        exception.InnerException.Should().BeNull();
    }

    [Fact]
    public void Constructor_WithMessageAndInnerException()
    {
        var innerException = new FormatException("The inner exception");
        var exception = new TheClassUnderTest("The message", innerException);

        exception.Message.Should().Be("The message");
        exception.InnerException.Should().BeSameAs(innerException);
    }
}
